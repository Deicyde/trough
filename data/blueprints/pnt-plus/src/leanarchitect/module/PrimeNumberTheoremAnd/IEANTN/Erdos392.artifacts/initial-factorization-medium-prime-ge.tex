\begin{sublemma}
\uses{params-set,initial-factorization-def,balance-def}

\leanok
A medium prime $\sqrt{n} < p ≤ n/L$ can be in deficit by at most $M$.
\end{sublemma}
\begin{proof}
\uses{factorization-def}
\leanok
The number of times $p$ divides $a_1 \dots a_t$ is at least $M \lfloor n/Mp
  \rfloor ≥ n/p - M$ (note that the removal of the non-smooth numbers
  does not remove any multiples
  of $p$).  Meanwhile, the number of times $p$ divides $n!$ is at most $n/p$.
\end{proof}
