\begin{definition}[RectangleIntegral]
\label{RectangleIntegral}
\lean{RectangleIntegral}
% at /home/runner/work/trough/trough/library/PrimeNumberTheoremAnd/ResidueCalcOnRectangles.lean:36.0-49.59
\leanok
A RectangleIntegral of a function $f$ is one over a rectangle
  determined by $z$ and $w$ in $\C$.
  We will sometimes denote it by $\int_{z}^{w} f$.
  (There is also a primed version, which is
  $1/(2\pi i)$ times the original.)
\end{definition}
