\begin{theorem}[HolomorphicOn-riemannZeta0]
\uses{riemannZeta0}
\label{HolomorphicOn_riemannZeta0}
\lean{HolomorphicOn_riemannZeta0}
% at /home/runner/work/trough/trough/library/PrimeNumberTheoremAnd/ZetaBounds.lean:1301.0-1312.92
\leanok
For any $N\ge1$, the function $\zeta_0(N,s)$ is holomorphic on $\{s\in \C\mid \Re(s)>0 ∧ s \ne 1\}$.
\end{theorem}
\begin{proof}
\leanok
The function $\zeta_0(N,s)$ is a finite sum of entire functions, plus an integral
  that's absolutely convergent on $\{s\in \C\mid \Re(s)>0 ∧ s \ne 1\}$ by Lemma \ref{ZetaBnd_aux1b}.
\end{proof}
