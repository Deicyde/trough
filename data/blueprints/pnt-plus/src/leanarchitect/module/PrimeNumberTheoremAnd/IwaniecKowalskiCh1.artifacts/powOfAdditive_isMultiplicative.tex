\begin{theorem}[powOfAdditive-isMultiplicative]
\uses{IsAdditive}
\label{powOfAdditive_isMultiplicative}
\lean{ArithmeticFunction.powOfAdditive_isMultiplicative}
% at /home/runner/work/trough/trough/library/PrimeNumberTheoremAnd/IwaniecKowalskiCh1.lean:954.0-975.70
\leanok
If $f$ is an additive function, then so to is $n\mapsto k^{f(n)}$ for any fixed $k$.
\end{theorem}
\begin{proof}
\leanok
Note that $k^{f(mn)}=k^{f(m)+f(n)}=k^{f(m)}k^{f(n)}$.
\end{proof}
