\begin{theorem}[sigmaR-one]
\uses{sigmaR}
\label{sigmaR_one}
\lean{ArithmeticFunction.sigmaR_one}
% at /home/runner/work/trough/trough/library/PrimeNumberTheoremAnd/IwaniecKowalskiCh1.lean:478.0-490.83
\leanok
We have that $\sigma^R_s(1)=1$.
\end{theorem}
\begin{proof}
\uses{sigmaR_apply}
\leanok
By definition we have that
    $$\sigma^R_s(1)=\sum_{d \mid 1}d^s=1^s=1.$$
\end{proof}
