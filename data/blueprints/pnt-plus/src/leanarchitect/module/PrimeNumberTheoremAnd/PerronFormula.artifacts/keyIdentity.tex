\begin{lemma}[keyIdentity]
\label{keyIdentity}
\lean{Perron.keyIdentity}
% at /home/runner/work/trough/trough/library/PrimeNumberTheoremAnd/PerronFormula.lean:811.0-825.53
\leanok
Let $x\in \R$ and $s \ne 0, -1$. Then
  $$
  \frac{x^\sigma}{s(1+s)} = \frac{x^\sigma}{s} - \frac{x^\sigma}{1+s}
  $$
\end{lemma}
\begin{proof}
\leanok
By ring.
\end{proof}
