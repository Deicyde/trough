\begin{lemma}[Preliminary decay bound III]
\label{prelim-decay-3}
\lean{prelim_decay_3}
\discussion{563}

If $\psi:\R \to \C$ is absolutely integrable, absolutely continuous, and $\psi'$ is of bounded
variation, then
$$ |\hat \psi(u)| \leq \| \psi' \|_{TV} / (2\pi |u|)^2$$
for all non-zero $u \in \R$.
\end{lemma}
