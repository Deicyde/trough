\begin{lemma}[LogDerivZetaBnd]
\label{LogDerivZetaBnd}
\lean{LogDerivZetaBnd}
% at /home/runner/work/trough/trough/library/PrimeNumberTheoremAnd/ZetaBounds.lean:2997.0-3044.84
\leanok
There is an $A>0$ so that for $1-A/\log^9 |t| \le \sigma < 1+A/\log^9 |t|$ and $3 < |t|$,
  $$
  |\frac {\zeta'}{\zeta} (\sigma+it)| \ll \log^9 |t|.
  $$
\end{lemma}
\begin{proof}
\uses{ZetaDerivUpperBnd,ZetaInvBnd}
\leanok
Combine the bound on $|\zeta'|$ from Lemma \ref{ZetaDerivUpperBnd} with the
  bound on $1/|\zeta|$ from Lemma \ref{ZetaInvBnd}.
\end{proof}
