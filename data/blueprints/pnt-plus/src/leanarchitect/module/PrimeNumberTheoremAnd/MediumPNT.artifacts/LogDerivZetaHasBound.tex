\begin{definition}[LogDerivZetaHasBound]
\label{LogDerivZetaHasBound}
\lean{LogDerivZetaHasBound}
% at /home/runner/work/trough/trough/library/PrimeNumberTheoremAnd/MediumPNT.lean:807.0-815.25
\leanok
For constants $A$, $C$, $n_1$, and $n_2$ this is true if for all $3<|t|$ one has
  $$1-\frac{A}{\log|t|^{n_1}}\leq\sigma\implies|(\zeta'/\zeta)(\sigma+it)|\leq C\,\log|t|^{n_2}.$$
\end{definition}
