    (hg : Dense (Subgroup.closure (Set.range g) : Set G)) :

      (∀ U : OpenNormalSubgroup G, finiteAugmentation _ (c U) = 0) →
      (∀ ⦃U V : OpenNormalSubgroup G⦄ (h : U ≤ V),
        quotientAlgebraMap Λ h (c U) - c V ∈ finiteAugmentationIdealSq Λ (G ⧸ (V : Subgroup G))) →
