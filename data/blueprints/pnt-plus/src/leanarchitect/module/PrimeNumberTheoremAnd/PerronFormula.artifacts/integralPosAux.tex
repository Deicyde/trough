\begin{lemma}[integralPosAux]
\label{integralPosAux}
\lean{Perron.integralPosAux}
% at /home/runner/work/trough/trough/library/PrimeNumberTheoremAnd/PerronFormula.lean:474.0-487.36
\leanok
The integral
  $$\int_\R\frac{1}{|(1+t^2)(2+t^2)|^{1/2}}dt$$
  is positive (and hence convergent - since a divergent integral is zero in Lean, by
  definition).
\end{lemma}
\begin{proof}
\leanok
This integral is between $\frac{1}{2}$ and $1$ of the integral of $\frac{1}{1+t^2}$,
  which is $\pi$.
\end{proof}
