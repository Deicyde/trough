\begin{sublemma}[limiting behavior of log zeta]
\label{log-zeta-limit}
\lean{_private.PrimeNumberTheoremAnd.IEANTN.Mertens.0.Mertens.log_zeta_limit}
\discussion{1586}
% at /home/runner/work/trough/trough/library/PrimeNumberTheoremAnd/IEANTN/Mertens.lean:1655.0-1684.56
\leanok
One has $\log \zeta(s) = - \log(s-1) + o(1)$ as $s \to 1^+$.
\end{sublemma}
\begin{proof}
\leanok
Start with the asymptotic $\zeta(s) = \frac{1}{s-1} + O(1)$ and take logarithms.
\end{proof}
