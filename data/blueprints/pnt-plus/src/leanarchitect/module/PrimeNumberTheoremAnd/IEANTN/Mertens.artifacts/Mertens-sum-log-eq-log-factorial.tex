\begin{proposition}[Partial sum of logarithm as logarithm of factorial]
\label{Mertens-sum-log-eq-log-factorial}
\lean{Mertens.sum_log_eq_log_factorial}
\discussion{1315}
% at /home/runner/work/trough/trough/library/PrimeNumberTheoremAnd/IEANTN/Mertens.lean:425.0-440.23
\leanok
For any $x \geq 1$, one has
$$ \sum_{n \leq x} \log n = \log(\lfloor x \rfloor!). $$
\end{proposition}
\begin{proof}
\leanok
Follows from the definition of the factorial function.  Is not needed for the Mertens theorems, but is a natural fact to have.
\end{proof}
