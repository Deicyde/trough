\begin{lemma}[tendsto-zero-Lower]
\label{tendsto_zero_Lower}
\lean{Perron.tendsto_zero_Lower}
% at /home/runner/work/trough/trough/library/PrimeNumberTheoremAnd/PerronFormula.lean:679.0-697.75
\leanok
Let $x>0$ and $\sigma',\sigma''\in\R$. Then
  $$\int_{\sigma'}^{\sigma''}\frac{x^{\sigma+it}}{(\sigma+it)(1+\sigma + it)}d\sigma$$
  goes to $0$ as $t\to-\infty$.
\end{lemma}
\begin{proof}
\leanok
The numerator is bounded and the denominator tends to infinity.
\end{proof}
