\begin{theorem}[moebius-sq-tsum-prime-pow]
\uses{sumOnPrimePows}
\label{moebius_sq_tsum_prime_pow}
\lean{ArithmeticFunction.moebius_sq_tsum_prime_pow}
% at /home/runner/work/trough/trough/library/PrimeNumberTheoremAnd/IwaniecKowalskiCh1.lean:1446.0-1482.28
\leanok
For $p$ prime, we have that
    $$\sum_{0\leq k}\mu^2(p^k)p^{-ks}=1+p^{-s}.$$
\end{theorem}
\begin{proof}
\uses{sumOnPrimePows_apply}
\leanok
Note that $\mu^2(p^0)=1$ and $\mu^2(p^1)=1$ but $\mu^2(p^k)=0$ whenever $1<k$. Thus,
    $$\sum_{0\leq k}\mu^2(p^k)p^{-ks}=1+p^{-s}.$$
\end{proof}
