\begin{definition}[ChebyshevPsi]
\label{ChebyshevPsi}
\lean{ChebyshevPsi}
% at /home/runner/work/trough/trough/library/PrimeNumberTheoremAnd/MediumPNT.lean:39.0-50.17
\leanok
The (second) Chebyshev Psi function is defined as
  $$
  \psi(x) := \sum_{n \le x} \Lambda(n),
  $$
  where $\Lambda(n)$ is the von Mangoldt function.
\end{definition}
