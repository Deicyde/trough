\begin{lemma}[ZetaInvBound1]
\label{ZetaInvBound1}
\lean{ZetaInvBound1}

For all $\sigma>1$,
  $$
  1/|\zeta(\sigma+it)| \le |\zeta(\sigma)|^{3/4}|\zeta(\sigma+2it)|^{1/4}
  $$
\end{lemma}
\begin{proof}
\leanok
The identity
  $$
  1 \le |\zeta(\sigma)|^3 |\zeta(\sigma+it)|^4 |\zeta(\sigma+2it)|
  $$
  for $\sigma>1$
  is already proved by Michael Stoll in the EulerProducts PNT file.
\end{proof}
