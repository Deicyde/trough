\begin{definition}[li and Li]
\label{li-def}
\lean{li,Li}
% at /home/runner/work/trough/trough/library/PrimeNumberTheoremAnd/Defs.lean:72.0-82.17
\leanok
$\mathrm{li}(x) = \int_0^x \frac{dt}{\log t}$ (in the
  principal value sense) and
  $\mathrm{Li}(x) = \int_2^x \frac{dt}{\log t}$.
\end{definition}
