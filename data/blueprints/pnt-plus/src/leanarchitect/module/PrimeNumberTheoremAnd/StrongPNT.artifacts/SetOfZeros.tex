\begin{definition}[SetOfZeros]
\label{SetOfZeros}
\lean{SetOfZeros}
% at /home/runner/work/trough/trough/library/PrimeNumberTheoremAnd/StrongPNT.lean:345.0-351.73
\leanok
Let $R>0$ and $f:\mathbb{C}\to\mathbb{C}$. Define the set of zeros
    $\mathcal{K}_f(R)=\{\rho\in\mathbb{C}:|\rho|\leq R,\,f(\rho)=0\}$.
\end{definition}
