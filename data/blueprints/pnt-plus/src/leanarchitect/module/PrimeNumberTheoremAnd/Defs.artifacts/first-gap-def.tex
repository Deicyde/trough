\begin{definition}[First prime gap]
\label{first-gap-def}
\lean{first_gap}
% at /home/runner/work/trough/trough/library/PrimeNumberTheoremAnd/Defs.lean:32.0-42.8
\leanok
$P(g)$ is the first prime $p_n$ for which the prime gap
  $p_{n+1}-p_n$ is equal to $g$, or $0$ if no such gap
  exists.
\end{definition}
