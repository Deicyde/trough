\begin{lemma}[finsum-range-eq-sum-range]
\label{finsum_range_eq_sum_range}
\lean{finsum_range_eq_sum_range}
% at /home/runner/work/trough/trough/library/PrimeNumberTheoremAnd/Consequences.lean:38.0-53.28
\leanok
For any arithmetic function $f$ and real number $x$, one has
  $$ \sum_{n \leq x} f(n) = \sum_{n \leq ⌊x⌋_+} f(n)$$
  and
  $$ \sum_{n < x} f(n) = \sum_{n < ⌈x⌉_+} f(n).$$
\end{lemma}
\begin{proof}
\leanok
Straightforward.
\end{proof}
