\begin{theorem}[SmoothExistence]
\label{SmoothExistence}
\lean{SmoothExistence}
% at /home/runner/work/trough/trough/library/PrimeNumberTheoremAnd/SmoothExistence.lean:50.0-107.22
\leanok
There exists a smooth (once differentiable would be enough),
  nonnegative ``bumpfunction'' $\nu$,
  supported in $[1/2,2]$ with total mass one:
  $$
  \int_0^\infty \nu(x)\frac{dx}{x} = 1.
  $$
\end{theorem}
\begin{proof}
\leanok
Same idea as Urysohn-type argument.
\end{proof}
