\begin{corollary}[Integral form for second error (von Mangoldt form)]
\uses{Mertens-second-error-mangoldt,Mertens-first-error-mangoldt}
\label{Mertens-second-error-mangoldt-eq}
\lean{Mertens.E₂Λ.eq}
\discussion{1317}
% at /home/runner/work/trough/trough/library/PrimeNumberTheoremAnd/IEANTN/Mertens.lean:1011.0-1047.84
\leanok
For any $x \geq 2$, one has
$$ E_{2,\Lambda}(x) = \frac{E_{1,\Lambda}(x)}{\log x} - \int_x^\infty \frac{E_{1,\Lambda}(t)}{t \log^2 t}\ dt$$
\end{corollary}
\begin{proof}
\uses{Euler-Mascheroni-const-alt,Mertens-integral-ident,Mertens-first-theorem-mangoldt}
\leanok
From Lemma \ref{Mertens-integral-ident} one has
$$ \sum_{n \leq x} \frac{\Lambda(n)}{n \log n} = \frac{1}{\log x} \sum_{n \leq x} \frac{\Lambda(n)}{n} + \int_2^x \frac{1}{t \log^2 t} \sum_{n \leq t} \frac{\Lambda(n)}{n} \, dt.$$
Now substitute the definitions of $E_{1,\Lambda}$, $E_{2,\Lambda}$, $\gamma$ and simplify.
\end{proof}
