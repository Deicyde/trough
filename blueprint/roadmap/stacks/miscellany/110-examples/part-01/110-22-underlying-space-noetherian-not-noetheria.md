theorem localization_away_x_equiv_laurentPolynomial :
    ∃ e : Localization.Away (x k) ≃+* LaurentPolynomial (Polynomial k),
      (∀ c : k,
        e (algebraMap (subalgebra k) (Localization.Away (x k))
          (algebraMap k (subalgebra k) c)) =
        LaurentPolynomial.C (Polynomial.C c)) ∧
      e (algebraMap (subalgebra k) (Localization.Away (x k)) (x k)) =
        LaurentPolynomial.T 1 ∧
      (∀ i : ℕ,
        e (algebraMap (subalgebra k) (Localization.Away (x k))
          (xyPowIn k i)) =
        LaurentPolynomial.T 1 * LaurentPolynomial.C (Polynomial.X ^ i))
