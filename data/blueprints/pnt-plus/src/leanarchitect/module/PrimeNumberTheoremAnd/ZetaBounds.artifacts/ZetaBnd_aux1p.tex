\begin{lemma}[ZetaBnd-aux1p]
\label{ZetaBnd_aux1p}
\lean{ZetaBnd_aux1p}
% at /home/runner/work/trough/trough/library/PrimeNumberTheoremAnd/ZetaBounds.lean:1057.0-1092.17
\leanok
For any $N\ge1$ and $s = \sigma + tI \in \C$, $\sigma=\in(0,2], 2 < |t|$,
  $$
  \left| s\int_N^\infty \frac{\lfloor x\rfloor + 1/2 - x}{x^{s+1}} \, dx \right|
  \ll |t| \frac{N^{-\sigma}}{\sigma}.
  $$
\end{lemma}
\begin{proof}
\uses{ZetaBnd_aux1b}
\leanok
Apply Lemma \ref{ZetaBnd_aux1b} and estimate $|s|\ll |t|$.
\end{proof}
