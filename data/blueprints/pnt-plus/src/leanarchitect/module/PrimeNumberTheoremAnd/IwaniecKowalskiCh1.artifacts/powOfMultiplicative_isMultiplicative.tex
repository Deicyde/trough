\begin{theorem}[powOfMultiplicative-isMultiplicative]
\label{powOfMultiplicative_isMultiplicative}
\lean{ArithmeticFunction.powOfMultiplicative_isMultiplicative}
% at /home/runner/work/trough/trough/library/PrimeNumberTheoremAnd/IwaniecKowalskiCh1.lean:1412.0-1428.52
\leanok
If $f$ is a multiplicative function, then so to is $n\mapsto f(n)^k$ for any $k\in\mathbb{N}$.
\end{theorem}
\begin{proof}
\leanok
Note that $f(mn)^k=(f(m)f(n))^k=(f(m)^k)(f(n)^k)$.
\end{proof}
