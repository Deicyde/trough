\begin{lemma}[Zeta-eq-int-derivZeta]
\label{Zeta_eq_int_derivZeta}
\lean{Zeta_eq_int_derivZeta}
% at /home/runner/work/trough/trough/library/PrimeNumberTheoremAnd/ZetaBounds.lean:2614.0-2648.65
\leanok
For any $t\ne0$ (so we don't pass through the pole), and $\sigma_1 < \sigma_2$,
  $$
  \int_{\sigma_1}^{\sigma_2}\zeta'(\sigma + it) dt =
  \zeta(\sigma_2+it) - \zeta(\sigma_1+it).
  $$
\end{lemma}
\begin{proof}
\leanok
This is the fundamental theorem of calculus.
\end{proof}
