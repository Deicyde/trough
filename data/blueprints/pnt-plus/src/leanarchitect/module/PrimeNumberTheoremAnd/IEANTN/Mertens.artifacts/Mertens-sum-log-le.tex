\begin{lemma}[Partial sum of logarithm upper bound]
\label{Mertens-sum-log-le}
\lean{Mertens.sum_log_le}
\discussion{1304}
% at /home/runner/work/trough/trough/library/PrimeNumberTheoremAnd/IEANTN/Mertens.lean:342.0-363.38
\leanok
For any $x \geq 1$, one has
$$ \sum_{n \leq x} \log n \leq x \log x.$$
\end{lemma}
\begin{proof}
\leanok
Trivial since $\log n \leq \log x$.
\end{proof}
