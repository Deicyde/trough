  $$
  \frac1{2\pi i}
  \int_{(-3/2)}\frac{x^s}{s(s+1)}ds = \frac 1{2\pi i}
  \int_{(\sigma)}\frac{x^s}{s(s+1)}ds.
  $$
\end{lemma}
