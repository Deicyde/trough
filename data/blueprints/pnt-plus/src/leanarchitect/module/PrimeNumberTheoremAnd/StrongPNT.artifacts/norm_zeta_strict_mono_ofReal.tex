\begin{theorem}[norm-zeta-strict-mono-ofReal]
\label{norm_zeta_strict_mono_ofReal}
\lean{norm_zeta_strict_mono_ofReal}
% at /home/runner/work/trough/trough/library/PrimeNumberTheoremAnd/StrongPNT.lean:1457.0-1491.85
\leanok
If $a,b\in\mathbb{R}$ with $1<b<a$, then
    $$|\zeta(a)|<|\zeta(b)|.$$
\end{theorem}
\begin{proof}
\leanok
Expand out $\zeta$ as a summation (we can do this as we are in the region of convergence) and
    ignore the norm as every term is positive and real. Comparing term by term proves the result.
\end{proof}
