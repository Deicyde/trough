\begin{theorem}[LSeries.term-isMultiplicative-if-fun-isMultiplicative]
\label{LSeries.term_isMultiplicative_if_fun_isMultiplicative}
\lean{ArithmeticFunction.LSeries.term_isMultiplicative_if_fun_isMultiplicative}
% at /home/runner/work/trough/trough/library/PrimeNumberTheoremAnd/IwaniecKowalskiCh1.lean:935.0-952.67
\leanok
If $f$ is a multiplicative function, then so to is $n\mapsto f(n)/n^s$.
\end{theorem}
\begin{proof}
\leanok
Note that $f(mn)/(mn)^s=f(m)f(n)/(m^sn^s)=(f(m)/m^s)(f(n)/n^s)$.
\end{proof}
