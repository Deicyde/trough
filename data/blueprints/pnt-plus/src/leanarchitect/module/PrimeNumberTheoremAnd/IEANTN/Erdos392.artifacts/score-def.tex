\begin{definition}
\uses{waste-def,factorization-def,balance-def}
\label{score-def}
\lean{Erdos392.Factorization.score}
% at /home/runner/work/trough/trough/library/PrimeNumberTheoremAnd/IEANTN/Erdos392.lean:122.0-136.42
\leanok
The score of a factorization (relative to a cutoff parameter $L$) is equal to its waste,
  plus $\log p$ for every surplus prime $p$, $\log (n/p)$ for every deficit prime above $L$,
  $\log L$ for every deficit prime below $L$ and an additional $\log n$ if one is not in
  total balance.
\end{definition}
