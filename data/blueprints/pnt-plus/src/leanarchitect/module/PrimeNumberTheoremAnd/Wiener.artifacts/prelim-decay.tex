\begin{lemma}[Preliminary decay bound I]
\label{prelim-decay}
\lean{prelim_decay}
\discussion{561}
% at /home/runner/work/trough/trough/library/PrimeNumberTheoremAnd/Wiener.lean:301.0-311.56
\leanok
If $\psi:\R \to \C$ is absolutely integrable then $$ |\hat \psi(u)| \leq \| \psi \|_1 $$
  for all $u \in \R$. where $C$ is an absolute constant.
\end{lemma}
\begin{proof}
\leanok
Immediate from the triangle inequality.
\end{proof}
