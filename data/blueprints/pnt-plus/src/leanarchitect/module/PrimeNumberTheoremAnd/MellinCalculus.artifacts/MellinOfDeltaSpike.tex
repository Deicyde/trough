\begin{theorem}[MellinOfDeltaSpike]
\uses{DeltaSpike}
\label{MellinOfDeltaSpike}
\lean{MellinOfDeltaSpike}
% at /home/runner/work/trough/trough/library/PrimeNumberTheoremAnd/MellinCalculus.lean:663.0-677.9
\leanok
For any $\epsilon>0$, the Mellin transform of $\nu_\epsilon$ is
  $$\mathcal{M}(\nu_\epsilon)(s) = \mathcal{M}(\nu)\left(\epsilon s\right).$$
\end{theorem}
\begin{proof}
\leanok
Substitute $y=x^{1/\epsilon}$, use Haar measure; direct calculation.
\end{proof}
