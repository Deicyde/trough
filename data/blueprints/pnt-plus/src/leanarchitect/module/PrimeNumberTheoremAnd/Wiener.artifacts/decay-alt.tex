\label{decay-alt}
\lean{decay_alt}
\discussion{564}

\leanok
If $\psi:\R \to \C$ is absolutely
integrable, absolutely continuous, and $\psi'$ is of bounded variation, then
$$ |\hat \psi(u)| \leq ( \|\psi\|_1 + \| \psi' \|_{TV} / (2\pi)^2) / (1+|u|^2)$$
for all $u \in \R$.
\end{lemma}
