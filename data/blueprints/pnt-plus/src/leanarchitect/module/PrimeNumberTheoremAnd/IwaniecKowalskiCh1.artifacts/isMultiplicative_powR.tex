\begin{theorem}[isMultiplicative-powR]
\uses{powR}
\label{isMultiplicative_powR}
\lean{ArithmeticFunction.isMultiplicative_powR}

\leanok
For fixed $\nu$ the function $n\mapsto n^\nu$ is multiplicative.
\end{theorem}
\begin{proof}
\leanok
This immediately follows from the fact that exponentiation with a fixed power is a homomorphism.
\end{proof}
