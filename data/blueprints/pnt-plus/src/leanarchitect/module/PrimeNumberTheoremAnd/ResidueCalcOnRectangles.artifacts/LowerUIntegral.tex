\begin{definition}[LowerUIntegral]
\label{LowerUIntegral}
\lean{LowerUIntegral}

A LowerUIntegral of a function $f$ comes from
  $\sigma-i\infty$ up to $\sigma-iT$, over to
  $\sigma'-iT$, and back down to $\sigma'-i\infty$.
\end{definition}
